import unicodedata

def normalize_text(text):
    if not isinstance(text, str):
        return ""
    # Normalize unicode (e.g., Ö -> O)
    text = unicodedata.normalize('NFD', text)
    text = "".join(c for c in text if unicodedata.category(c) != 'Mn')
    text = re.sub(r'[^a-zA-Z0-9]', '', text)
    return text.lower()

# Prepare participants
part_list = []
for idx, row in df_part.iterrows():
    cog = str(row['Cognome']).strip() if pd.notna(row['Cognome']) else ""
    nom = str(row['Nome']).strip() if pd.notna(row['Nome']) else ""
    email = str(row['Email']).strip() if pd.notna(row['Email']) else ""
    
    cog_norm = normalize_text(cog)
    nom_norm = normalize_text(nom)
    
    part_list.append({
        'cognome': cog,
        'nome': nom,
        'email': email,
        'cog_norm': cog_norm,
        'nom_norm': nom_norm,
        'comb_norm': cog_norm + nom_norm
    })

# Process nomifile
matched_count = 0
unmatched = []

output_list = []

for idx, file_path in enumerate(df_files[0].dropna()):
    file_path_str = str(file_path).strip()
    basename = os.path.basename(file_path_str)
    basename_norm = normalize_text(basename)
    
    # Try matching
    best_match = None
    best_score = 0
    
    for p in part_list:
        # Check if both cognome and nome are in basename
        if p['cog_norm'] and p['nom_norm']:
            if p['cog_norm'] in basename_norm and p['nom_norm'] in basename_norm:
                score = len(p['cog_norm']) + len(p['nom_norm'])
                if score > best_score:
                    best_score = score
                    best_match = p
        elif p['cog_norm'] and p['cog_norm'] in basename_norm:
            score = len(p['cog_norm'])
            if score > best_score:
                best_score = score
                best_match = p

    if not best_match:
        # Try matching only cog_norm if cog_norm >= 3 chars
        for p in part_list:
            if p['cog_norm'] and len(p['cog_norm']) >= 3 and p['cog_norm'] in basename_norm:
                score = len(p['cog_norm'])
                if score > best_score:
                    best_score = score
                    best_match = p

        matched_count += 1
        output_list.append({
            'nomefile': file_path_str,
            'email': best_match['email'],
            'cognome': best_match['cognome'],
            'nome': best_match['nome']
        })
    else:
        unmatched.append((file_path_str, basename_norm))

print(f"Total files: {len(df_files[0].dropna())}")

def escape_yaml_str(val):
    if val is None:
        return ""
    val = str(val).strip()
    # Se contiene caratteri speciali, facciamo il quote
    if any(c in val for c in [':', '{', '}', '[', ']', ',', '&', '*', '#', '?', '|', '-', '<', '>', '=', '!', '%', '@', '\\', '"']):
        # Escape quotes
        val_escaped = val.replace('\\', '\\\\').replace('"', '\\"')
        return f'"{val_escaped}"'
    return val

yaml_filename = 'mappatura_partecipanti.yaml'

Non ho accesso diretto ai contenuti dei file allegati per eseguirli automaticamente in questo ambiente. Tuttavia, ecco lo script Python che puoi eseguire per incrociare i due file CSV e generare il file YAML richiesto:

```python
import csv
import yaml

# 1. Carica i partecipanti in un dizionario basato sul cognome (in minuscolo per confronto sicuro)
partecipanti = {}
with open('__partecipanti.csv', mode='r', encoding='utf-8') as f:
    reader = csv.DictReader(f)
    for row in reader:
        # Pulisce eventuali spazi bianchi attorno ai nomi delle colonne e ai valori
        row = {k.strip(): v.strip() for k, v in row.items()}
        cognome = row['cognome'].strip()
        partecipanti[cognome.lower()] = {
            'nome': row['nome'],
            'cognome': cognome,
            'email': row['email']
        }

# 2. Leggi i nomi dei file e associa i dati del partecipante corrispondente
risultato = []
with open('__nomifile.csv', mode='r', encoding='utf-8') as f:
    reader = csv.DictReader(f)
    # Assumi che la colonna con il nome del file si chiami 'nomefile'
    colonna_file = reader.fieldnames[0] 
    
    for row in reader:
        nome_file = row[colonna_file].strip()
        
        # Cerca quale cognome è presente nel nome del file
        match_trovato = False
        for cognome_lower, dati in partecipanti.items():
            if cognome_lower in nome_file.lower():
                risultato.append({
                    'nomefile': nome_file,
                    'email': dati['email'],
                    'cognome': dati['cognome'],
                    'nome': dati['nome']
                })
                match_trovato = True
                break

# 3. Scrivi il risultato nel file YAML
with open('output.yaml', mode='w', encoding='utf-8') as f:
    yaml.dump(risultato, f, sort_keys=False, allow_unicode=True)

print("File YAML creato con successo!")

```
